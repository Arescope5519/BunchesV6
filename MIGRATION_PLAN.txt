================================================================================
GOAL
================================================================================
Prevent duplicate recipes by implementing a global recipe repository.
- Recipes imported from URLs are stored once globally
- Users reference the global version and have their own local edits/tags
- Maintains data integrity while allowing personalization

================================================================================
CURRENT ARCHITECTURE
================================================================================
Table: recipes
- Each user has full copies of every recipe they save
- Duplicate data when multiple users save the same URL
- All fields stored per-user (title, ingredients, instructions, etc.)

================================================================================
TARGET ARCHITECTURE
================================================================================

Table: global_recipes (NEW)
----------------------------------------
- id (uuid, primary key)
- source_url (text, UNIQUE) - the canonical URL
- title (text)
- ingredients (jsonb)
- instructions (jsonb)
- image_url (text)
- prep_time (text)
- cook_time (text)
- total_time (text)
- servings (text)
- nutrition (jsonb)
- created_at (timestamp)
- updated_at (timestamp)

Notes:
- Immutable once created (represents the "original" recipe)
- Keyed by source_url to prevent duplicates
- Any user importing same URL references this record


Table: user_recipes_v2 (NEW)
----------------------------------------
- id (uuid, primary key)
- user_id (uuid, references auth.users)
- global_recipe_id (uuid, references global_recipes) - NULL for manual recipes
- local_recipe_data (jsonb) - for manual recipes without URL, stores full recipe
- local_edits (jsonb) - user's modifications to global recipe
- folder (text)
- tags (text[])
- is_favorite (boolean)
- notes (text)
- deleted_at (timestamp)
- created_at (timestamp)
- updated_at (timestamp)

Notes:
- If global_recipe_id is set, recipe came from URL (reference global)
- If global_recipe_id is NULL, local_recipe_data contains full manual recipe
- local_edits contains user's customizations (merged with global on read)

================================================================================
IMPORT FLOW (After Migration)
================================================================================

1. User shares URL to app
2. App extracts/parses recipe data
3. Check: Does global_recipes have this source_url?

   YES ->
     - Get existing global_recipe_id
     - Create user_recipes_v2 entry pointing to it
     - User sees the recipe in their cookbook

   NO ->
     - Insert new record into global_recipes
     - Create user_recipes_v2 entry pointing to new global record
     - User sees the recipe in their cookbook

4. User makes edits:
   - Edits stored in user_recipes_v2.local_edits
   - Global recipe remains unchanged
   - User can toggle between "original" and "my edits"


================================================================================
MANUAL RECIPE FLOW (No URL)
================================================================================

1. User creates recipe manually (no source URL)
2. Insert into user_recipes_v2 with:
   - global_recipe_id = NULL
   - local_recipe_data = full recipe JSON
3. No global_recipes entry created
4. Recipe is user-specific only


================================================================================
SHARING FLOW (After Migration)
================================================================================

1. User A shares recipe with User B
2. If recipe has global_recipe_id:
   - Share the global_recipe_id reference
   - User B gets their own user_recipes_v2 entry pointing to same global
3. If recipe is manual (no global_recipe_id):
   - Share the local_recipe_data
   - User B gets copy in their local_recipe_data


================================================================================
DATA TO MIGRATE

-- 1. Create global_recipes table
CREATE TABLE IF NOT EXISTS global_recipes (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  source_url text UNIQUE NOT NULL,
  title text,
  ingredients jsonb,
  instructions jsonb,
  image_url text,
  prep_time text,
  cook_time text,
  total_time text,
  servings text,
  nutrition jsonb,
  author text,
  description text,
  cuisine text,
  category text,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now()
);

-- 2. Create user_recipes_v2 table
CREATE TABLE IF NOT EXISTS user_recipes_v2 (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id uuid REFERENCES auth.users(id) ON DELETE CASCADE NOT NULL,
  global_recipe_id uuid REFERENCES global_recipes(id) ON DELETE SET NULL,
  local_recipe_data jsonb,
  local_edits jsonb,
  folder text DEFAULT 'All Recipes',
  tags text[] DEFAULT '{}',
  is_favorite boolean DEFAULT false,
  notes text,
  imported_from text,
  imported_at timestamp with time zone,
  deleted_at timestamp with time zone,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),

  -- Ensure either global_recipe_id OR local_recipe_data is set
  CONSTRAINT recipe_source_check CHECK (
    global_recipe_id IS NOT NULL OR local_recipe_data IS NOT NULL
  )
);

-- 3. Create indexes for performance
CREATE INDEX IF NOT EXISTS idx_global_recipes_source_url ON global_recipes(source_url);
CREATE INDEX IF NOT EXISTS idx_user_recipes_v2_user_id ON user_recipes_v2(user_id);
CREATE INDEX IF NOT EXISTS idx_user_recipes_v2_global_recipe_id ON user_recipes_v2(global_recipe_id);

-- 4. Enable RLS
ALTER TABLE global_recipes ENABLE ROW LEVEL SECURITY;
ALTER TABLE user_recipes_v2 ENABLE ROW LEVEL SECURITY;

-- 5. RLS Policies for global_recipes
-- Anyone can read global recipes
CREATE POLICY "Anyone can read global recipes"
  ON global_recipes FOR SELECT
  TO authenticated
  USING (true);

-- Authenticated users can insert global recipes
CREATE POLICY "Authenticated users can insert global recipes"
  ON global_recipes FOR INSERT
  TO authenticated
  WITH CHECK (true);

-- 6. RLS Policies for user_recipes_v2
-- Users can only access their own recipes
CREATE POLICY "Users can access own recipes v2"
  ON user_recipes_v2 FOR ALL
  TO authenticated
  USING (auth.uid() = user_id)
  WITH CHECK (auth.uid() = user_id);

-- 7. Enable realtime for new tables
ALTER PUBLICATION supabase_realtime ADD TABLE user_recipes_v2;

Migration SQL: READY TO RUN

-- STEP 1: Migrate URL-based recipes to global_recipes
-- Takes the first occurrence of each unique URL
INSERT INTO global_recipes (
  source_url,
  title,
  ingredients,
  instructions,
  image_url,
  prep_time,
  cook_time,
  servings,
  author,
  created_at,
  updated_at
)
SELECT DISTINCT ON (source_url)
  source_url,
  title,
  CASE
    WHEN ingredients IS NULL THEN NULL
    WHEN ingredients::text LIKE '{%' OR ingredients::text LIKE '[%' THEN ingredients::jsonb
    ELSE to_jsonb(ingredients)
  END as ingredients,
  CASE
    WHEN instructions IS NULL THEN NULL
    WHEN instructions::text LIKE '[%' THEN instructions::jsonb
    ELSE to_jsonb(instructions)
  END as instructions,
  image_url,
  prep_time,
  cook_time,
  servings,
  NULL as author,
  created_at,
  NOW() as updated_at
FROM recipes
WHERE source_url IS NOT NULL
  AND source_url != ''
  AND NOT EXISTS (
    SELECT 1 FROM global_recipes g WHERE g.source_url = recipes.source_url
  )
ORDER BY source_url, created_at ASC;

-- STEP 2: Migrate manual recipes (no URL) to global_recipes
-- Uses bunches://user/{user_id}/recipes/{recipe_id} as source
INSERT INTO global_recipes (
  source_url,
  title,
  ingredients,
  instructions,
  image_url,
  prep_time,
  cook_time,
  servings,
  author,
  created_at,
  updated_at
)
SELECT
  'bunches://user/' || user_id || '/recipes/' || id as source_url,
  title,
  CASE
    WHEN ingredients IS NULL THEN NULL
    WHEN ingredients::text LIKE '{%' OR ingredients::text LIKE '[%' THEN ingredients::jsonb
    ELSE to_jsonb(ingredients)
  END as ingredients,
  CASE
    WHEN instructions IS NULL THEN NULL
    WHEN instructions::text LIKE '[%' THEN instructions::jsonb
    ELSE to_jsonb(instructions)
  END as instructions,
  image_url,
  prep_time,
  cook_time,
  servings,
  'Bunches User' as author,
  created_at,
  NOW() as updated_at
FROM recipes
WHERE (source_url IS NULL OR source_url = '')
  AND NOT EXISTS (
    SELECT 1 FROM global_recipes g
    WHERE g.source_url = 'bunches://user/' || recipes.user_id || '/recipes/' || recipes.id
  );

-- STEP 3: Create user_recipes_v2 entries for URL-based recipes
INSERT INTO user_recipes_v2 (
  id,
  user_id,
  global_recipe_id,
  local_recipe_data,
  local_edits,
  folder,
  tags,
  is_favorite,
  notes,
  deleted_at,
  created_at,
  updated_at
)
SELECT
  r.id,
  r.user_id,
  g.id as global_recipe_id,
  NULL as local_recipe_data,
  CASE
    WHEN r.has_edits = true AND r.edited_version IS NOT NULL THEN
      jsonb_build_object(
        'variants', jsonb_build_array(
          jsonb_build_object(
            'id', 'migrated-edit',
            'name', 'My Edits',
            'edits', r.edited_version,
            'createdAt', extract(epoch from r.updated_at) * 1000
          )
        ),
        'selectedVariantId', CASE WHEN r.viewing_original = true THEN NULL ELSE 'migrated-edit' END
      )
    WHEN r.variants IS NOT NULL AND jsonb_array_length(r.variants) > 0 THEN
      jsonb_build_object('variants', r.variants, 'selectedVariantId', r.selected_variant_id)
    ELSE NULL
  END as local_edits,
  COALESCE(r.folder, 'All Recipes') as folder,
  COALESCE(r.tags, ARRAY[]::text[]) as tags,
  COALESCE((r.folder = 'Favorites'), false) as is_favorite,
  r.notes,
  r.deleted_at,
  r.created_at,
  NOW() as updated_at
FROM recipes r
JOIN global_recipes g ON g.source_url = r.source_url
WHERE r.source_url IS NOT NULL
  AND r.source_url != ''
  AND NOT EXISTS (
    SELECT 1 FROM user_recipes_v2 u WHERE u.id = r.id
  );

-- STEP 4: Create user_recipes_v2 entries for manual recipes
INSERT INTO user_recipes_v2 (
  id,
  user_id,
  global_recipe_id,
  local_recipe_data,
  local_edits,
  folder,
  tags,
  is_favorite,
  notes,
  deleted_at,
  created_at,
  updated_at
)
SELECT
  r.id,
  r.user_id,
  g.id as global_recipe_id,
  NULL as local_recipe_data,
  CASE
    WHEN r.variants IS NOT NULL AND jsonb_array_length(r.variants) > 0 THEN
      jsonb_build_object('variants', r.variants, 'selectedVariantId', r.selected_variant_id)
    ELSE NULL
  END as local_edits,
  COALESCE(r.folder, 'All Recipes') as folder,
  COALESCE(r.tags, ARRAY[]::text[]) as tags,
  COALESCE((r.folder = 'Favorites'), false) as is_favorite,
  r.notes,
  r.deleted_at,
  r.created_at,
  NOW() as updated_at
FROM recipes r
JOIN global_recipes g ON g.source_url = 'bunches://user/' || r.user_id || '/recipes/' || r.id
WHERE (r.source_url IS NULL OR r.source_url = '')
  AND NOT EXISTS (
    SELECT 1 FROM user_recipes_v2 u WHERE u.id = r.id
  );
